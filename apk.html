// ডাটাবেজ থেকে ডেটা এনে লেআউট রেন্ডার করার পারফেক্ট ফাংশন function renderDatabaseLayout(company) { let fullHtml = ""; const settings = company.layout_settings; // ১. সেটিংস অবজেক্ট বা জেসন স্ট্রিং যাই হোক না কেন তা থেকে ফুল HTML আলাদা করা if (typeof settings === 'object' && settings !== null) { fullHtml = settings.full_html || settings.custom_html || ""; } else if (typeof settings === 'string') { try { let parsed = JSON.parse(settings); fullHtml = parsed.full_html || parsed.custom_html || settings; } catch(ex) { fullHtml = settings; // যদি সরাসরি প্লেন এইচটিএমএল টেক্সট থাকে } } if (!fullHtml) { alert("ডেটাবেজের layout_settings কলামে কোনো কার্যকর HTML কোড পাওয়া যায়নি!"); return; } // ২. সিগমা পিএইচডির মূল ফ্রেম হাইড করে ফুলস্ক্রিন হোল্ডার ওপেন করা document.getElementById('sigmaMainWrapper').classList.add('hidden'); const fullscreenHolder = document.getElementById('fullScreenCompanyHolder'); fullscreenHolder.classList.remove('hidden'); // ৩. আল নাওয়া কোম্পানির ফুল HTML এখানে ইনজেকশন করা fullscreenHolder.innerHTML = fullHtml; // ৪. ভেতরের সকল জাভাস্ক্রিপ্ট ও স্ক্রিপ্ট ট্যাগ সিকিউরলি এক্সিকিউট করা const scripts = fullscreenHolder.querySelectorAll("script"); scripts.forEach(oldScript => { const newScript = document.createElement("script"); Array.from(oldScript.attributes).forEach(attr => newScript.setAttribute(attr.name, attr.value)); if (oldScript.src) { newScript.src = oldScript.src; } else { newScript.appendChild(document.createTextNode(oldScript.innerHTML)); } oldScript.parentNode.replaceChild(newScript, oldScript); }); }
